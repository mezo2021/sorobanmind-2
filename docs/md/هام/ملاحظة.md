## 1. تقييم التصميم الحالي (بالأدلة)

- `wipeLocalProgress` (`cloudSync.ts` L234–262) **يحذف** كل مفتاح `cloud:*` يطابق `OUR_PREFIX` (L226) ثم يعيد ضبط الـ store. أي أن بادئة الوضع تجعل كل حسابات Cloud تشترك في مساحة `cloud:` واحدة تُمسح عند التبديل: `cloud:sorobanmind-v2-progress`، `cloud:soroban_high_scores_*`، وأي مفتاح جديد من P2-03/04/05.

- التقدم الرئيسي يعود للحساب العائد من Supabase فقط إن نجح `saveToCloud` عند الخروج.

**الخلاصة:** بادئة الوضع وحدها **لا تكفي**، والحذف الحالي مخالف للشرط. إعادة تسمية مفاتيح كل شاشة بـuid تحل ثلاث شاشات فقط وتترك الـ store والدرجات العالية والمفاتيح الخمسة، وتمس 8+ ملفات ← **مرفوض**.

## 2. التصميم الموصى به: «خزنة الحساب» (Account Vault)

**الفكرة:** المفاتيح **النشطة** (`cloud:<base>` والخمسة) تخص الحساب الحالي فقط. عند تبديل الحساب تُنقل بدل أن تُحذف إلى خزنة باسم المالك: `cloud:<uid>:<base>`، وتُستعاد خزنة الحساب القادم إلى المفاتيح النشطة. فلا يرى ج بيانات ب (المفاتيح النشطة تُفرَّغ)، ولا يضيع شيء. **لا تغيير في الشاشات ولا في `storage.ts`.**

**القواعد (يُنفَّذ داخل `adoptUser` وتحلّ محل حذف `wipeLocalProgress`):**
1. **المالك السابق A موجود و≠ الجديد B:** (أ) انسخ كل مفتاح نشط (`cloud:*` المطابق `OUR_PREFIX` + المفاتيح الخمسة) إلى `cloud:A:<base>` وتحقق بالقراءة، ثم أزل النشط؛ ومفاتيح خزنة A التي لم تعد لها نظير نشط تُزال (استبدال خزنة A نفسها هو الحذف الوحيد المسموح). (ب) انسخ `cloud:B:<base>` إن وُجدت إلى النشطة. (ج) كالسابق: إعادة ضبط الـ store وتطبيق تقدم السحابة.
2. **لا مالك سابق:** تُنقل المفاتيح النشطة إلى خزنة `cloud:__unowned__:<base>` (تُحفظ ولا تُستعاد أبداً) بدل حذفها.
3. **المالك نفسه:** لا شيء (LOCAL WINS).

3. وضّح كيف يشمل Rollback تغييرات الخزنة نفسها، وليس المفاتيح النشطة فقط. المطلوب ضمان سلامة البيانات عند فشل أي خطوة أو انقطاع التطبيق في منتصف العملية.

4. تحقق من تعريف "OUR_PREFIX" ومطابقته الفعلية للمفاتيح المقترحة، خصوصاً "cloud:srb_progress" و"cloud:soroban_lesson_session_<id>" و"cloud:soroban_secrets_best_scores". لا تفترض أنها كلها مطابقة دون دليل.

// خزنة الحساب: نقل بيانات الحساب بين المفاتيح النشطة والخزنة بلا حذف، مع سجل معاملة للاسترداد.
export interface KV {
  readonly length: number;
  key(index: number): string | null;
  getItem(key: string): string | null;
  setItem(key: string, value: string): void;
  removeItem(key: string): void;
}

export const OUR_PREFIX = /^(soroban|sorobanmind|srb|sorobania)/;
export const UNOWNED = '__unowned__';
export const ACCOUNT_BOUND_UNSCOPED_KEYS = [
  'sorobania_progress_v2',
  'sorobania_progress_v1',
  'soroban_friends_weak_v1',
  'sorobanmind-ghost-v1',
  'soroban_companion',
] as const;

// أسماء الميتا لا تطابق OUR_PREFIX عمداً (كي لا يمسحها Snapshot)
const SLOT_KEY = 'vault:slot';
const TXN_KEY = 'vault:txn';
const UUID = '[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}';
const VAULT_RE = new RegExp(`^cloud:(?:${UUID}|${UNOWNED}):`, 'i');

export function isVaultKey(key: string): boolean {
  return key.startsWith('vault:') || VAULT_RE.test(key);
}

interface Txn { from: string; to: string; step: 'out' | 'in' }
const txnJson = (t: Txn): string => JSON.stringify(t);

const isFive = (k: string): boolean =>
  (ACCOUNT_BOUND_UNSCOPED_KEYS as readonly string[]).includes(k);

/** اسم المفتاح داخل الخزنة، أو null إن لم يكن من بيانات الحساب */
function baseOf(key: string): string | null {
  if (key.startsWith('cloud:')) {
    if (isVaultKey(key)) return null;
    const bare = key.slice(6);
    return OUR_PREFIX.test(bare) ? bare : null;
  }
  return isFive(key) ? key : null;
}
const activeKeyOf = (base: string): string => (isFive(base) ? base : `cloud:${base}`);
const vaultKeyOf = (uid: string, base: string): string => `cloud:${uid}:${base}`;
const prevKeyOf = (uid: string, base: string): string => `cloud:${uid}:@prev:${base}`;
const recKeyOf = (uid: string, base: string): string => `cloud:${uid}:@rec:${base}`;

function writeVerified(s: KV, key: string, value: string): void {
  s.setItem(key, value);
  if (s.getItem(key) !== value) throw new Error(`vault: write verify failed: ${key}`);
}
function removeVerified(s: KV, key: string): void {
  s.removeItem(key);
  if (s.getItem(key) !== null) throw new Error(`vault: remove verify failed: ${key}`);
}

function allKeys(s: KV): string[] {
  const out: string[] = [];
  for (let i = 0; i < s.length; i++) {
    const k = s.key(i);
    if (k) out.push(k);
  }
  return out;
}

/** [مفتاح نشط، base، قيمة] */
function listActive(s: KV): Array<[string, string, string]> {
  const out: Array<[string, string, string]> = [];
  for (const k of allKeys(s)) {
    const base = baseOf(k);
    if (base === null) continue;
    const v = s.getItem(k);
    if (v !== null) out.push([k, base, v]);
  }
  return out;
}

/** [base، قيمة] لخزنة uid (بدون @prev/@rec) */
function listVault(s: KV, uid: string): Array<[string, string]> {
  const prefix = `cloud:${uid}:`;
  const out: Array<[string, string]> = [];
  for (const k of allKeys(s)) {
    if (!k.startsWith(prefix)) continue;
    const base = k.slice(prefix.length);
    if (base.startsWith('@')) continue;
    const v = s.getItem(k);
    if (v !== null) out.push([base, v]);
  }
  return out;
}

/** نسخ إلى الخزنة بلا حذف؛ الاستبدال يحفظ القيمة السابقة في @prev */
function copyToVault(s: KV, uid: string, base: string, value: string): void {
  const vk = vaultKeyOf(uid, base);
  const old = s.getItem(vk);
  if (old !== null && old !== value) writeVerified(s, prevKeyOf(uid, base), old);
  writeVerified(s, vk, value);
}

function parseTxn(raw: string): Txn {
  let t: unknown;
  try { t = JSON.parse(raw); } catch { throw new Error('vault: corrupt txn'); }
  const o = t as Partial<Txn> | null;
  if (!o || typeof o.from !== 'string' || typeof o.to !== 'string' || (o.step !== 'out' && o.step !== 'in')) {
    throw new Error('vault: corrupt txn');
  }
  return { from: o.from, to: o.to, step: o.step };
}

export function getSlot(s: KV): string | null { return s.getItem(SLOT_KEY); }
export function hasPendingTxn(s: KV): boolean { return s.getItem(TXN_KEY) !== null; }

/** هل يجب اعتماد الحساب قبل عرض بياناته؟ (مالك/خانة/معاملة معلقة) */
export function needsAdoption(s: KV, userId: string, owner: string | null): boolean {
  if (hasPendingTxn(s)) return true;
  const slot = getSlot(s) ?? owner ?? UNOWNED;
  return owner !== userId || slot !== userId;
}

/** استرداد معاملة معلقة: يعيد دائماً إلى حالة «الخانة = from» أو يُكمل ما اكتمل */
export function recoverVault(s: KV): 'none' | 'completed' | 'rolled-back' {
  const raw = s.getItem(TXN_KEY);
  if (raw === null) return 'none';
  const txn = parseTxn(raw);

  if (txn.step === 'in' && getSlot(s) === txn.to) {
    removeVerified(s, TXN_KEY);          // النسخ اكتمل وتحقق قبل تحديث الخانة
    return 'completed';
  }
  if (txn.step === 'in') {
    for (const [k, base, v] of listActive(s)) {
      if (s.getItem(vaultKeyOf(txn.to, base)) === v) removeVerified(s, k);   // نسخة من خزنة to
      else {                                                                  // غريب: يُحفظ ولا يُفقد
        writeVerified(s, recKeyOf(txn.from, base), v);
        removeVerified(s, k);
      }
    }
  }
  for (const [base, v] of listVault(s, txn.from)) {
    const ak = activeKeyOf(base);
    if (s.getItem(ak) === null) writeVerified(s, ak, v);     // لا نكتب فوق نشط موجود
  }
  writeVerified(s, SLOT_KEY, txn.from);
  removeVerified(s, TXN_KEY);
  return 'rolled-back';
}

/** مبادلة: بيانات صاحب الخانة ← خزنته، وخزنة userId ← النشط. لا حذف من أي خزنة. */
export function vaultSwap(
  s: KV,
  userId: string,
  legacyOwner: string | null,
): { status: 'same' | 'swapped'; from: string } {
  recoverVault(s);
  let slot = getSlot(s);
  if (slot === null) {
    slot = legacyOwner || UNOWNED;
    writeVerified(s, SLOT_KEY, slot);
  }
  if (slot === userId) return { status: 'same', from: slot };
  const from = slot;

  writeVerified(s, TXN_KEY, txnJson({ from, to: userId, step: 'out' }));

  const copied = new Map<string, string>();
  for (const [k, base, v] of listActive(s)) {
    copyToVault(s, from, base, v);
    copied.set(k, v);
  }
  for (const [k, v] of copied) {
    const cur = s.getItem(k);
    if (cur !== null && cur !== v) copyToVault(s, from, baseOf(k) as string, cur); // تغيّر أثناء النقل
    removeVerified(s, k);
  }
  for (const [k, base, v] of listActive(s)) {   // مفاتيح ظهرت أثناء النقل
    copyToVault(s, from, base, v);
    removeVerified(s, k);
  }

  writeVerified(s, TXN_KEY, txnJson({ from, to: userId, step: 'in' }));
  for (const [base, v] of listVault(s, userId)) writeVerified(s, activeKeyOf(base), v);

  writeVerified(s, SLOT_KEY, userId);
  removeVerified(s, TXN_KEY);
  return { status: 'swapped', from };
}
