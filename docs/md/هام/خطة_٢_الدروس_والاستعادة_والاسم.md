**ب) الدوال الثلاث.** استبدل الدوال `loadSession` و`saveSession` و`clearSession` (الثلاث متتالية) بما يلي:
```ts
function sessionKey(lessonId: string): string | null {
  const mode = getMode();
  return mode ? `${mode}:${LESSON_SESSION_PREFIX}${lessonId}` : null;
}

function loadSession(lessonId: string): SessionData {
  try {
    const key = sessionKey(lessonId);
    if (!key) return {};
    const raw = localStorage.getItem(key);
    return raw ? JSON.parse(raw) : {};
  } catch {
    return {};
  }
}

function saveSession(lessonId: string, data: SessionData): void {
  try {
    const key = sessionKey(lessonId);
    if (!key) return;
    localStorage.setItem(key, JSON.stringify(data));
  } catch { /* ignore */ }
}

function clearSession(lessonId: string): void {
  try {
    const key = sessionKey(lessonId);
    if (!key) return;
    localStorage.removeItem(key);
  } catch { /* ignore */ }
}
```
(النص القديم الذي يُستبدل يستعمل `LESSON_SESSION_PREFIX + lessonId` مباشرة.)

**الملفان:** `src/lib/cloudSync.ts` و`src/screens/GuardianDashboard.tsx` (PR واحد).

### أ) `cloudSync.ts`

**1) تصدير الدوال.** غيّر ثلاثة أسطر بإضافة `export`:

- `interface LocalStateSnapshot {` ← `export interface LocalStateSnapshot {`
- `function createLocalStateSnapshot(): LocalStateSnapshot {` ← `export function createLocalStateSnapshot(): LocalStateSnapshot {`
- `function restoreLocalStateSnapshot(snapshot: LocalStateSnapshot): void {` ← `export function restoreLocalStateSnapshot(snapshot: LocalStateSnapshot): void {`

**2) استبدل الدالة `async loadFromCloud() { ... },` كاملة** (من سطر `  async loadFromCloud() {` حتى سطر `  },` الذي يسبق `  async submitScore(` مباشرة) بـ:
```ts
  async loadFromCloud() {
    requireCloudMode();

    const user = await this.getCurrentUser();
    if (!user) throw new CloudSyncError('Must be logged in', 'AUTH_REQUIRED');

    const { data, error } = await supabase
      .from('soroban_progress')
      .select('progress_data')
      .eq('user_id', user.id)
      .maybeSingle();

    if (error) {
      throw new CloudSyncError('Failed to fetch cloud progress', 'PROGRESS_FETCH_FAILED');
    }

    const pd = data?.progress_data as unknown;
    // لا صف أو null أو غير كائن: نرفض قبل أي تغيير. الكائن الفارغ {} صالح.
    if (!pd || typeof pd !== 'object' || Array.isArray(pd)) {
      throw new CloudSyncError('No valid cloud backup found', 'PROGRESS_FETCH_FAILED');
    }

    const { data: profile, error: profileError } = await supabase
      .from('public_profiles')
      .select('avatar_id')
      .eq('user_id', user.id)
      .maybeSingle();

    if (profileError) {
      throw new CloudSyncError('Failed to fetch account profile', 'PROFILE_FETCH_FAILED');
    }

    const { childName: _c, parentName: _p, ...safePd } = pd as Record<string, unknown>;

    // reset() يعيد اسم الطفل واسم ولي الأمر إلى الفارغ، والنسخة السحابية لا تحملهما
    const keepNames = {
      childName: useProgressStore.getState().childName,
      parentName: useProgressStore.getState().parentName,
    };

    useProgressStore.getState().reset(); // استبدال كامل لا دمج
    useProgressStore.setState({ ...(safePd as any), ...keepNames });

    if (profile?.avatar_id) {
      localStorage.setItem('soroban_hero_key', profile.avatar_id);
    }

    applyLocalIdentity();
    return true;
  },
```
لا تعدّل `pullProgress` ولا `adoptUser` ولا `saveToCloud`.

### ب) `GuardianDashboard.tsx`

**2) الدالة.** استبدل `const handleCloudLoad = async () => { ... };` كاملة بـ:
```ts
  const restoringRef = useRef(false);
  const [isRestoring, setIsRestoring] = useState(false);

  const handleCloudLoad = async () => {
    if (restoringRef.current) return;
    const confirmed = window.confirm(
      pickLang(
        {
          ar: 'سيتم استبدال تقدمك المحلي ببيانات السحابة. هل تريد المتابعة؟',
          en: 'Your local progress will be replaced by cloud data. Continue?',
        },
        lang,
      ),
    );
    if (!confirmed) return;

    restoringRef.current = true;
    setIsRestoring(true);
    let snapshot: LocalStateSnapshot | null = null;
    try {
      snapshot = createLocalStateSnapshot();
      await cloudSync.loadFromCloud();
      setCloudMsg({
        text: pickLang({ ar: 'تم الاستعادة ☁️ جاري التحديث...', en: 'Restored ☁️ Reloading...' }, lang),
        type: 'success',
      });
      setTimeout(() => window.location.reload(), 1200);
    } catch (e: any) {
      let rollbackFailed = false;
      if (snapshot) {
        try {
          restoreLocalStateSnapshot(snapshot);
        } catch {
          rollbackFailed = true;
        }
      }
      setCloudMsg({
        text: rollbackFailed
          ? pickLang(
              {
                ar: 'فشلت الاستعادة وتعذّر التراجع؛ حالة البيانات غير مؤكدة. حاول مجدداً.',
                en: 'Restore failed and rollback failed; data state is uncertain. Try again.',
              },
              lang,
            )
          : (e?.message ?? 'Restore failed'),
        type: 'error',
      });
      setTimeout(() => setCloudMsg(null), 3000);
    } finally {
      restoringRef.current = false;
      setIsRestoring(false);
    }
  };
```
