س4: لماذا تغير تصميم الخصوصية؟

ج: بعد نقاش مطول مع مساعدين AI ومناقشات قانونية (COPPA/GDPR)، اكتشفنا أن:

2. عرض اسم الطفل علناً يُعد "إفصاحاً عن بيانات شخصية".
3. حفظ اسم ولي الأمر الثلاثي في user_metadata ليس آمناً (قابل للتعديل من المستخدم).

س5: ما التصميم الجديد؟

ج: استراتيجية "تقليل البيانات" (Data Minimization):

البيانات مكان الحفظ السبب

كلمة المرور السحابة (مشفرة) للمصادقة
اسم الطفل (مفرد "جود") محلي فقط (localStorage) للشهادات المحلية
اسم ولي الأمر (أي عدد كلمات) محلي فقط للشهادات المحلية
عمر الطفل محلي فقط للفئة الدراسية
اللقب (حيوان ثابت) السحابة للوحات الشرف
الدولة (اختيارية) السحابة للوحات
الموافقات الثلاث السحابة (user_metadata) للامتثال القانوني

س10: ما هي مخاطر التصميم الجديد؟

ج: (من مراجعة كلاود وGPT):

🧪 الجزء التاسع: SQL النهائي المنفذ فعلاً

```sql
-- ═══════════════════════════════════════════════════════
-- 1) جدول الألعاب
-- ═══════════════════════════════════════════════════════
CREATE TABLE public.score_games (
  game_id TEXT PRIMARY KEY,
  max_score INT NOT NULL CHECK (max_score > 0)
);

-- (24 صفاً: 8 مستويات + 2 أقسام + 7 كيو + 7 ألعاب)

-- ═══════════════════════════════════════════════════════
-- 2) جدول الملفات الشخصية
-- ═══════════════════════════════════════════════════════
CREATE TABLE public.public_profiles (
  user_id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  display_name TEXT NOT NULL DEFAULT 'بطل',
  show_on_leaderboard BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- ═══════════════════════════════════════════════════════
-- 3) جدول الدرجات
-- ═══════════════════════════════════════════════════════
CREATE TABLE public.scores (
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  game_id TEXT NOT NULL REFERENCES public.score_games(game_id),
  period_key TEXT NOT NULL CHECK (period_key = 'all' OR period_key ~ '^\d{4}-(0[1-9]|1[0-2])$'),
  score INT NOT NULL CHECK (score >= 0),
  time_ms INT CHECK (time_ms IS NULL OR time_ms >= 0),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  PRIMARY KEY (user_id, game_id, period_key)
);

CREATE INDEX scores_top ON public.scores (game_id, period_key, score DESC, time_ms ASC NULLS LAST);

-- ═══════════════════════════════════════════════════════
-- 4) دالة mask_name
-- ═══════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION public.mask_name(p TEXT) RETURNS TEXT
LANGUAGE sql IMMUTABLE AS $$
  SELECT CASE WHEN array_length(w, 1) >= 2
              THEN w[1] || '.' || LEFT(w[array_length(w, 1)], 1)
              ELSE COALESCE(NULLIF(w[1], ''), 'بطل') END
  FROM (SELECT regexp_split_to_array(TRIM(COALESCE(p, '')), '\s+') AS w) t;
$$;

-- ═══════════════════════════════════════════════════════
-- 5) Trigger محمي
-- ═══════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION public.handle_new_profile() RETURNS TRIGGER
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
BEGIN
  BEGIN
    INSERT INTO public.public_profiles(user_id, display_name)
    VALUES (NEW.id, public.mask_name(NEW.raw_user_meta_data ->> 'child_name'))
    ON CONFLICT DO NOTHING;
  EXCEPTION WHEN OTHERS THEN NULL;
  END;
  RETURN NEW;
END $$;

DROP TRIGGER IF EXISTS on_auth_user_profile ON auth.users;
CREATE TRIGGER on_auth_user_profile AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_profile();

-- ═══════════════════════════════════════════════════════
-- 6) submit_score
-- ═══════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION public.submit_score(
  p_game_id TEXT, p_score INT, p_time_ms INT DEFAULT NULL
) RETURNS VOID LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
DECLARE 
  v_uid UUID := auth.uid(); 
  v_max INT; 
  v_key TEXT;
BEGIN
  IF v_uid IS NULL THEN RAISE EXCEPTION 'Not authenticated'; END IF;
  SELECT max_score INTO v_max FROM score_games WHERE game_id = p_game_id;
  IF v_max IS NULL OR p_score < 0 OR p_score > v_max THEN RAISE EXCEPTION 'Invalid score'; END IF;
  
  FOREACH v_key IN ARRAY ARRAY['all', to_char(timezone('UTC', now()), 'YYYY-MM')] LOOP
    INSERT INTO scores(user_id, game_id, period_key, score, time_ms)
    VALUES (v_uid, p_game_id, v_key, p_score, p_time_ms)
    ON CONFLICT (user_id, game_id, period_key) DO UPDATE
      SET score = EXCLUDED.score, time_ms = EXCLUDED.time_ms, updated_at = NOW()
      WHERE EXCLUDED.score > scores.score
         OR (EXCLUDED.score = scores.score AND COALESCE(EXCLUDED.time_ms, 2147483647) < COALESCE(scores.time_ms, 2147483647));
  END LOOP;
END $$;

-- ═══════════════════════════════════════════════════════
-- 7) get_hall_of_fame
-- ═══════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION public.get_hall_of_fame(
  p_game_id TEXT, p_period TEXT DEFAULT 'all', p_limit INT DEFAULT 10
) RETURNS TABLE(pos INT, display_name TEXT, score INT, time_ms INT, is_me BOOLEAN)
LANGUAGE sql SECURITY DEFINER SET search_path = public STABLE AS $$
  SELECT 
    (row_number() OVER (ORDER BY s.score DESC, s.time_ms ASC NULLS LAST, s.updated_at ASC))::INT,
    p.display_name, s.score, s.time_ms, (s.user_id = auth.uid())
  FROM scores s
  JOIN public_profiles p ON p.user_id = s.user_id AND p.show_on_leaderboard
  WHERE s.game_id = p_game_id
    AND s.period_key = CASE WHEN p_period = 'month' THEN to_char(timezone('UTC', now()), 'YYYY-MM') ELSE 'all' END
  ORDER BY s.score DESC, s.time_ms ASC NULLS LAST, s.updated_at ASC
  LIMIT LEAST(GREATEST(p_limit, 1), 50);
$$;

-- ═══════════════════════════════════════════════════════
-- 8) get_my_rank_details
-- ═══════════════════════════════════════════════════════
CREATE OR REPLACE FUNCTION public.get_my_rank_details(
  p_game_id TEXT, p_period TEXT DEFAULT 'all'
) RETURNS TABLE(game_id TEXT, period_key TEXT, player_rank BIGINT, score INTEGER, time_ms INTEGER)
LANGUAGE sql STABLE SECURITY DEFINER SET search_path = ''
AS $$
  WITH ranked_players AS (
    SELECT s.user_id, s.game_id, s.period_key, s.score, s.time_ms,
      ROW_NUMBER() OVER (ORDER BY s.score DESC, s.time_ms ASC NULLS LAST, s.updated_at ASC) AS player_rank
    FROM public.scores AS s
    JOIN public.public_profiles AS p ON p.user_id = s.user_id AND p.show_on_leaderboard
    WHERE s.game_id = p_game_id
      AND s.period_key = CASE WHEN p_period = 'month' THEN to_char(timezone('UTC', now()), 'YYYY-MM') ELSE 'all' END
  )
  SELECT r.game_id, r.period_key, r.player_rank, r.score, r.time_ms
  FROM ranked_players AS r
  WHERE r.user_id = auth.uid()
  LIMIT 1;
$$;
```
